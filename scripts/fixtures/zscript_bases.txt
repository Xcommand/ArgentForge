class Frost_MonsterBase : Actor abstract
{
	Default
	{
		Monster;
		DamageFactor "FrostDamage", 0;
		+FloorClip;

		+DONTHARMSPECIES;
		+DONTHARMCLASS;
	}
	override Void Tick()
	{
		super.Tick();
		{
			if(WaveToggle == 1)
			{
				if(!FindInventory("WaveHide",FALSE))
				{
					if(!self.bCORPSE)
					{	
						Spawn("Frost_Step1",Vec3Offset (0,0,32));
					}
				}
			}
		}
	}
	override Void Die(Actor source, Actor inflictor, int dmgflags, Name MeansOfDeath)
	{
		super.Die(source, inflictor, dmgflags, MeansOfDeath);
		{
			if(WaveToggle == 1)
			{
				Spawn("Frost_Step1Fade",Vec3Offset (0,0,32));
			}
		}
	}
}
